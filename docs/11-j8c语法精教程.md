## 第 1 课：变量与类型

变量 = 给数字/数据起个名字的盒子。用之前要先声明它的类型。

```c
void main() {
    u32 x = 5;          // 无符号 32 位整数（0 ~ 40 亿）
    i32 y = -7;         // 有符号 32 位整数
    u64 big = 10000000000;   // 更大的整数
    f64 f = 3.5;        // 小数（浮点）
    bool b = true;      // 真/假
    u8 c = 65;          // 小整数（也可以存字符）
    ptr p = 0;          // 指针（地址），0 = 空
}
```

| 类型 | 中文名 | 大小 | 范围 |
|---|---|---|---|
| `u8` / `u16` / `u32` / `u64` | 无符号整数 | 1/2/4/8 字节 | 0 起 |
| `i8` / `i16` / `i32` / `i64` | 有符号整数 | 1/2/4/8 字节 | 有负数 |
| `f64` | 小数 | 8 字节 | 有小数点 |
| `bool` | 布尔 | 1 字节 | true/false |
| `char` | 字符 | 1 字节 | 如 'A' |
| `ptr` / `T*` | 指针 | 8 字节 | 存地址 |

### ⚠️ 最重要的一个坑：整数字面量默认无符号

```c
u32 a = 7;
u32 b = a - 10;     // 结果不是 -3，而是 4294967293（u32 回绕！）
i32 c = 7;
i32 d = c - 10;     // 这样才是 -3
```

- `7`、`10` 这类**没有符号标记的字面量按无符号算**
- 要负数：写 `-7`（解析器特殊处理成有符号），或者变量声明成 `i32`/`i64`

## 第 2 课：运算符

### 算术

```c
void main() {
    print(1 + 2 * 3);     // 7（先乘除后加减）
    print(10 / 3);        // 3（整数除法，丢掉小数）
    print(10 % 3);        // 1（取余数）
    print(2 << 4);        // 32（左移 = 乘 2 的幂）
    print(255 & 15);      // 15（按位与）
    print(3.5 * 2.0);     // 7.000
}
```

### 比较与逻辑

```c
void main() {
    print(1 < 2);         // 1（true）
    print(5 == 5);        // 1
    print(5 != 5);        // 0
    print(1 < 2 && 3 > 2); // 1（&& 且）
    print(1 < 2 || 3 < 2); // 1（|| 或）
    print(!true);         // 0（! 非）
    u32 x = 5;
    print(x >= 5 ? 100 : 0); // 100（三元：条件 ? 真值 : 假值）
}
```

### 自增、复合赋值

```c
u32 i = 0;
i = i + 1;      // 或 i += 1;
i++;            // 或 ++i;
i *= 2;         // i = i * 2
```

## 第 3 课：控制流

### if / else

```c
void main() {
    i32 score = 85;
    if (score >= 90) {
        print("A");
    } else if (score >= 60) {
        print("B");
    } else {
        print("C");
    }
}
```

### while / do-while / for

```c
void main() {
    // while：先判断再循环
    u32 i = 0;
    while (i < 5) {
        print(i);
        i = i + 1;
    }
    // for：计数循环（初始化; 条件; 步进）
    u64 sum = 0;
    for (u32 j = 1; j <= 100; j = j + 1) {
        sum = sum + j;
    }
    print(sum);        // 5050
    // break / continue
    u32 k = 0;
    while (true) {     // 死循环
        k = k + 1;
        if (k == 3) continue;   // 跳过这次
        if (k >= 5) break;      // 跳出循环
        print(k);
    }
}
```

### return

```c
i32 abs2(i32 x) {
    if (x < 0) return -x;
    return x;
}
```

## 第 4 课：函数

```c
// 返回类型 函数名(参数类型 参数名, ...) { ... }
u64 fib(u64 n) {
    if (n < 2) return n;
    return fib(n - 1) + fib(n - 2);   // 递归：函数调用自己
}

u32 add3(u32 a, u32 b, u32 c) { return a + b + c; }

void say(u32 v) {           // void = 无返回值
    print(v);
}

void main() {
    print(fib(10));        // 55
    print(add3(1, 2, 3));  // 6
    say(777);
}
```

要点：
- 参数按值传递（复制一份进去，改不影响外面）
- 递归深度很安全（默认栈 1MB，能到上万层）
- **`return 字面量` 必须和返回类型匹配**——`u64 f() { return 0; }` 没问题，但类型别乱写

## 第 5 课：数组与指针

### 数组

```c
void main() {
    u8 buf[16];         // 16 个 u8 的数组
    buf[0] = 7;
    buf[1] = 9;
    print(buf[0]);      // 7
    print(buf[1]);      // 9
    u32 i = 0;
    while (i < 2) { print(buf[i]); i = i + 1; }
}
```

### 指针

```c
struct Point { u32 x; u32 y; }

void main() {
    Point p = { 10, 20 };     // 结构体初始化
    Point* q = &p;            // & = 取地址
    print(q->x);              // 10（-> = 通过指针访问成员）
    print(p.y);               // 20（. = 直接访问成员）
    ptr raw = malloc(64);     // 堆上分配 64 字节
    memcpy(raw, "hi", 3);     // 拷贝数据进去
    free(raw);                // 用完释放
}
```

## 第 8 课：面向协议

协议 = 约定一组方法；结构体说"我遵守这个协议"，就必须实现这些方法。

```c
// 定义协议（方法必须以 this 开头）
protocol Area {
    f64 calc(this);
}

// 实现协议（方法里用 this->字段 访问结构体字段）
struct Square { u32 w; }
impl Area for Square {
    f64 calc(this) { return (f64)this->w * (f64)this->w; }
}

struct Circle { f64 r; }
impl Area for Circle {
    f64 calc(this) { return 3.14159 * this->r * this->r; }
}

void main() {
    Square sq = { 5 };
    Circle ci = { 2.0 };
    print(sq.calc());   // 25.000
    print(ci.calc());   // 12.566
}
```

协议方法也可以有**默认实现**（`;` 换成 `{ ... }`），impl 可以不重写。

## 第 9 课：ECS（实体组件系统）

游戏里管理一堆对象的方式：**实体**（一个编号）+ **组件**（挂在实体上的数据）。

```c
// 定义组件
component Position { f64 x; f64 y; }
component Velocity { f64 vx; f64 vy; }

// 定义系统（处理一类实体）
system Move : Position, Velocity {
    fn update(ent e) {
        // e 是实体编号；get<组件>(e) 返回组件指针，用 -> 访问字段
        get<Position>(e)->x = get<Position>(e)->x + get<Velocity>(e)->vx;
        get<Position>(e)->y = get<Position>(e)->y + get<Velocity>(e)->vy;
    }
}

void main() {
    ent e = spawn();                    // 创建实体
    add<Position>(e, 0.0, 0.0);         // 挂上组件（字段值按顺序作参数）
    add<Velocity>(e, 1.0, 0.5);
    print(has<Position>(e));            // true（有）
    print(get<Position>(e)->x);         // 0.000
    remove<Velocity>(e);                // 摘掉组件
    despawn(e);                         // 销毁实体
    print(entity_count());              // 0
}
```

常用操作：`spawn` / `despawn` / `alive(e)` / `entity_count()` /
`add<T>(e, 字段值...)` / `remove<T>` / `has<T>` / `get<T>`（返回指针，`->` 访问）/ `run<S>`。

> ✅ `run<System>()` 已实现（原已知限制已解除）：`run<S>()` 遍历所有存活实体，
> 对满足系统全部组件的实体调用 `sys_S_update(e)`。示例：定义 `system Physics: Pos` 后
> 调 `run<Physics>()`，所有带 `Pos` 组件的实体的 `update` 都会跑一遍。
> （t5_ecs 回归测试覆盖：run 后组件值确实变化，18/18 全绿。）

## 第 10 课：外部函数与动态链接（最酷的部分）

J8 能直接调用 **C 语言的函数**，还能在运行时加载第三方库。

### extern：声明一个 C 函数

```c
extern i32 tp_add(i32, i32);   // 声明：lib 里有个这样的函数

void main() {
    print(tp_add(20, 22));     // 42（需要 --lib 预加载）
}
```

### 运行时动态链接（加载任意第三方库）

```c
void main() {
    ptr h  = dload("mylib.so");        // 1) 从 lib/ 目录加载动态库
    ptr fn = dlsym(h, "my_func");      // 2) 找到函数
    u32 id = dl_reg(fn, "i32(i32,i32)"); // 3) 注册（告诉 VM 签名）
    i32 r  = dl_call(id, "i32(i32,i32)", 1, 2); // 4) 调用
    print(r);
}
```

- `dload("x.so")` 自动去 `lib/` 目录找（= `dlopen("lib/x.so")`）
- **签名串**格式：`返回类型(参数类型,...)`，如 `"f64(f64,f64)"`、`"void(ptr)"`
- `getsystem()` 返回平台信息（低 16 位：1=Linux 2=Windows 3=macOS），可以按平台加载不同的库
- 方块跑酷游戏就是这么调用 SDL2 和 spdlog 的

## 第 11 课：综合练习——猜数字

把学到的都用上：

```c
extern i32 rand();          // C 库的随机数（需要 --lib libc 或已加载）
extern void srand(u32);

void main() {
    srand(12345);           // 固定随机种子
    u32 target = (u32)rand() % 100;
    u32 guess = 50;
    i32 tries = 0;
    while (true) {
        tries = tries + 1;
        if (guess == target) {
            print("found");
            print(tries);
            break;
        }
        if (guess < target) guess = guess + 1;
        else guess = guess - 1;
    }
}
```

## 附录：常见错误速查

| 报错/现象 | 原因 |
|---|---|
| `cannot assign 'u32' to 'void'` | 类型写错了，函数返回类型和 `return` 不匹配 |
| 负数算出来是巨大正数 | 整数字面量默认无符号，用 `i32`/`i64` 或 `-7` 写法 |
| `call to unknown function` | 函数没定义/没声明，或拼写错了 |
| `dl_call expects...` | 签名串必须用字符串字面量 |
| 窗口黑屏 | 物体颜色和背景色相同（改颜色数字） |
| `x * y;` 报错 | 被当成指针声明了（这是语言的已知歧义） |
